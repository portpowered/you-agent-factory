cmake_minimum_required(VERSION 3.20)
project(localai_qwen3_asr_bridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
set(QWEN3_ASR_TEST OFF CACHE BOOL "" FORCE)
if(QWEN3_ASR_BUILD)
  add_library(qwen3-asr SHARED IMPORTED)
  set_target_properties(qwen3-asr PROPERTIES IMPORTED_IMPLIB ${QWEN3_ASR_BUILD}/Release/qwen3-asr.lib)
else()
  add_subdirectory(${QWEN3_ASR_SOURCE} native)
endif()
add_library(goqwen3asrcpp SHARED bridge.cpp)
target_include_directories(goqwen3asrcpp PRIVATE ${QWEN3_ASR_SOURCE}/include ${QWEN3_ASR_SOURCE}/src ${QWEN3_ASR_SOURCE}/ggml/include)
target_link_libraries(goqwen3asrcpp PRIVATE qwen3-asr)
target_compile_definitions(goqwen3asrcpp PRIVATE NOMINMAX)
add_executable(qwen3-asr-seam-test seam_test.cpp)
add_executable(qwen3-asr-wave-reader-test wave_reader_test.cpp)
